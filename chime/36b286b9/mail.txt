template<template <typename> class M>
struct monad 
{ 
    template <class T, class Func>
    static auto bind(M<T>& m, Func&& func) -> result_type_t<Func>;
};

	==> monad is a meta function that takes a Template M and returns a type (a struct) (monad<M> is some struct type)
	==> monad<M>::bind is a meta function that takes two types T and F and returns a function (monad<M>::bind<T,F> is a function)
		==> result_type is a meta function that takes a type F and return a type (result_type<F> is the return type of function monad<M>::bind<T,F>)
