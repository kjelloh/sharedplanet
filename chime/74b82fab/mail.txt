#include <iostream>
#include <boost/bind.hpp>
#include <boost/function.hpp>

// Y-combinator compatible factorial
int fact(boost::function<int(int)> int_to_int,int v) {
    if(v == 0) {
        // Terminate
        return 1;
    }
    else {
        // Recurse
        return v * int_to_int(v -1);
    }
}

// Y-combinator for the int type
boost::function<int(int)>
y(boost::function<int(boost::function<int(int)>,int)> f_and_int_to_int) {
    return boost::bind(f_and_int_to_int,boost::bind(y,f_and_int_to_int),_1);
}

int main(int argc,char** argv) {
    
    {
        // std::cout << y(fact)(5) << std::endl;
    }
    
    {
        boost::function<int(int)> factorial = y(fact);
        std::cout << factorial(5) << std::endl;
    }

    return 0;
}
