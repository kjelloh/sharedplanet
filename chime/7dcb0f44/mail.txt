template <typename T, size_t N>

		==> This seems to mean that static_element_count takes a fixed sized raw C parameter by reference?

	==> Raw arrays always has a static capacity (declared at compile time)
	==> A raw array T[N] (some type T and element count N) seems to have type signature "T (&pX)[N]”?
	==> And we can specialise a template function on this type to capture raw array size at compile time

namespace raw_1 {
  
  struct A {
    int x;
    int y;
  };
  
  void test_and_print() {
    int i[3];
    float f[5];
    A a[7];
    std::cout << '\n' << "test_and_print output..." << '\n';
    std::cout << '\n' << "int i[3] \t has capacity for " << static_element_count(i) << " int{} objects";
    std::cout << '\n' << "float f[5] \t has capacity for " << static_element_count(f) << " float{} objects";
    std::cout << '\n' << "A a[7] \t\t has capacity for " << static_element_count(a) << " A{} objects";
    std::cout << '\n' << "\"Hello\"\t\t has capacity for " << static_element_count("Hello") << " char{} objects";
  }
  
}

namespace raw_4 {
  // Inspired by https://stackoverflow.com/questions/1461432/what-is-array-decaying <https://stackoverflow.com/questions/1461432/what-is-array-decaying>

  // "const T array[]"  same type as "const T* array"
  template <typename T> void by_value_1(const T* array) {
    std::cout << '\n' << "template <typename T> void by_value_1(const T* array) called.";
    std::cout << '\n' << "\tarray is T*, element count unknown";
    std::cout << '\n' << "\tarray = \"" << array << '\"';
    //std::cout << '\n' << "static_element_count(array) = " << static_element_count(array); // Error -  No matching function for call to 'static_element_count'
    std::cout << '\n' << "\tstatic_element_count(array); // Error -  No matching function for call to 'static_element_count'";
    std::cout << '\n' << "\tsizeof(array) = " << sizeof(array) << " = memory byte count (including allignment)";
  }

  // "const T array[]"  same type as "const T* array"
  template <typename T> void by_value_2(const T array[]) {
    std::cout << '\n' << "template <typename T> void by_value_1(const T* array) called.";
    std::cout << '\n' << "\tarray is T*, element count unknown";
    std::cout << '\n' << "\tarray = \"" << array << '\"';
    //std::cout << '\n' << "static_element_count(array) = " << static_element_count(array); // Error -  No matching function for call to 'static_element_count'
    std::cout << '\n' << "\tstatic_element_count(array); // Error -  No matching function for call to 'static_element_count'";
    std::cout << '\n' << "\tsizeof(array) = " << sizeof(array) << " = memory byte count (including allignment)";
  }

  template <typename T, size_t N> void by_pointer(const T (*array)[N]) {
    std::cout << '\n' << "\ttemplate <typename T, size_t N> void by_pointer(const T (*array)[N]) called";

  }

  template <typename T, size_t N> void by_reference(const T (&array)[N]) {
    std::cout << '\n' << "\ttemplate <typename T, size_t N> void by_reference(const T (&array)[N]) called";
    std::cout << '\n' << "\tarray = \"" << array << '\"';
    std::cout << '\n' << "\tstatic_element_count(array) = " << static_element_count(array) << " ok";
