	==> This page (UI Automation Client Programmer Guide) seem to be a good starting point (https://msdn.microsoft.com/en-us/library/ee684021(v=vs.85).aspx <https://msdn.microsoft.com/en-us/library/ee684021(v=vs.85).aspx>  )?

> Obtain an instance of the CUIAutomation object.
#include <uiautomation.h>
// CoInitialize must be called before calling this function, and the  
// caller must release the returned pointer when finished with it.
// 
HRESULT InitializeUIAutomation(IUIAutomation **ppAutomation)
{    
	return CoCreateInstance(
	   CLSID_CUIAutomation
	  ,NULL
          ,CLSCTX_INPROC_SERVER
	  ,IID_IUIAutomation
	  ,reinterpret_cast<void**>(ppAutomation));
}
