/*
 Model: Persistent Data Structure with Value semantics
 Msg: variant of message types
 ViewMsg: Implements Client Action -> Op Msg
 
 */
template <typename Model, typename Msg, typename ViewAndMsg>
class ElmRunTimeModel {
public:
    struct Sub { Msg msg;bool empty=true;operator bool() {return !empty;}};
    struct Cmd { Msg msg;bool empty=true;operator bool() {return !empty;}};
    struct Op { Msg msg;bool empty=true;operator bool() {return !empty;}};
    struct ModelAndMsg {Model model;Msg msg;};
    struct ModelAndCmd {Model model;Cmd cmd;};
    struct ModelAndSub {Model model;Sub cmd;};
    
    using SubscriptionsFn = std::function<ModelAndSub(const Model& model)>;
    using UpdateFn = std::function<ModelAndCmd(const ModelAndMsg& model_and_msg)>;
    using ViewFn = std::function<ViewAndMsg(const Model& model)>;
    
    ElmRunTimeModel(SubscriptionsFn subscriptions_fn, UpdateFn update_fn, ViewFn view_fn)
    : subscriptions_fn(subscriptions_fn), update_fn(update_fn), view_fn(view_fn) {}
    
    Model run(const ModelAndMsg& init_model_and_msg, Msg quit_msg) {
        auto current_model_and_msg = init_model_and_msg;
        
        while (true) {
            auto next_model_and_cmd = this->update_fn(current_model_and_msg);
            while (next_model_and_cmd.cmd) {
                if (next_model_and_cmd.cmd.msg == quit_msg) {
                    return next_model_and_cmd.model;
                }
                else{
                    current_model_and_msg = ModelAndMsg{next_model_and_cmd.model, next_model_and_cmd.cmd.msg};
                    next_model_and_cmd = this->update_fn(current_model_and_msg);
                }
            }
            auto view_and_msg = this->view_fn(current_model_and_msg.model);
            current_model_and_msg = ModelAndMsg{current_model_and_msg.model,post_to_clients(view_and_msg)};
        }
        return current_model_and_msg.model;
    }
    
private:
    SubscriptionsFn subscriptions_fn;
    UpdateFn update_fn;
    ViewFn view_fn;
    Msg post_to_clients(const ViewAndMsg& view_and_msg) {}
};
