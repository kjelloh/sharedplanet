// MyCppStringPlayground.cpp : Defines the entry point for the console application.

#include "stdafx.h"
#include <iostream>


int main()
{
	auto myNullTerminatedUnicodeString = u8"VÄLKOMMEN";
	auto iter = &myNullTerminatedUnicodeString[0];
	std::cout << "*iter type is " << typeid(*iter).name() << "\n";
	std::cout << "*iter size is " << sizeof(*iter) << "\n";
	std::cout << "myNullTerminatedUnicodeString = \"" << myNullTerminatedUnicodeString << "\"\n"; // cout will fail to interpret UTF-8

	std::cout << "\n** Listing *iter values **" << "\n";
	std::cout << "<char>" << '\t' << "<signed>" << '\t' << "<unsigned>" << "\n";
	do {
		std::cout << *iter << '\t' << static_cast<int>(*iter) << "\t\t" << static_cast<int>(static_cast<unsigned char>(*iter)) << "\n";
		++iter;
	} while (*iter != 0);

    return 0;
}
